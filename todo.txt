Use Vue Router's route guards: Same app, same build, but the admin UI is entirely isolated by layout and route guards.
// router/index.js
{
  path: '/office',
  component: AdminLayout,
  meta: { requiresAdmin: true },
  children: [ ... ]
}

router.beforeEach((to, from, next) => {
  if (to.meta.requiresAdmin && !store.getters.isAdmin) {
    next('/403')
  } else {
    next()
  }
})
